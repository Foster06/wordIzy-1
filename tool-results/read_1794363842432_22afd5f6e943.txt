   143→   143→/** Build the SEO title + description for a programmatic word-list config. */
   144→   144→function buildWordListSeo(config: WordListConfig): { title: string; description: string } {
   145→   145→  const dict = config.dict ?? "scrabble";
   146→   146→  const letter = String(config.value).toUpperCase();
   147→   147→
   148→   148→  // Resolve real word counts from the (module-cached) dictionary store.
   149→   149→  const slug = buildCanonicalWordListUrl(config).replace(/^\//, "");
   150→   150→  const data = getInitialWordListPage(slug, "en", 1);
   151→   151→  const fmt = (n: number) => n.toLocaleString("en-US");
   152→   152→  const total = data?.total ?? 0;
   153→   153→  const count5 = data?.lengthCounts?.[5] ?? 0;
   154→   154→
   155→   155→  if (config.type === "length") {
   156→   156→    const n = config.value as number;
   157→   157→    return {
   158→   158→      title: total > 0
   159→   159→        ? `${n}-Letter Words — ${fmt(total)} Valid Words`
   160→   160→        : `${n}-Letter Words — Complete Scrabble & Wordle List`,
   161→   161→      description: total > 0
   162→   162→        ? `Complete list of ${fmt(total)} ${n}-letter words from the official Scrabble dictionary, sorted by score.`
   163→   163→        : `Complete list of ${n}-letter words from the official Scrabble dictionary, sorted by score.`,
   164→   164→    };
   165→   165→  }
   166→   166→
   167→   167→  const starts = config.type === "starts";
   168→   168→
   169→   169→  if (dict === "wordle") {
   170→   170→    return {
   171→   171→      title: count5 > 0
   172→   172→        ? `${fmt(count5)} 5-Letter Words ${starts ? "Starting" : "Ending"} With "${letter}" — Wordle`
   173→   173→        : `Wordle Words That ${starts ? "Start" : "End"} With "${letter}"`,
   174→   174→      description: starts
   175→   175→        ? `${count5 > 0 ? `${fmt(count5)} five-letter ` : "All "}Wordle words that start with "${letter}", plus 6-7 letter lists. Sorted by score.`
   176→   176→        : `${count5 > 0 ? `${fmt(count5)} five-letter ` : "All "}Wordle words that end with "${letter}". Find hooks, suffixes & patterns fast.`,
   177→   177→    };
   178→   178→  }
   179→   179→
   180→   180→  return {
   181→   181→    title: total > 0
   182→   182→      ? `Words That ${starts ? "Start" : "End"} With "${letter}" — ${fmt(total)} Valid Words`
   183→   183→      : `Words That ${starts ? "Start" : "End"} With "${letter}" — Scrabble Word List`,
   184→   184→    description: starts
   185→   185→      ? `${total > 0 ? `${fmt(total)} valid ` : "All "}words that start with "${letter}" in the official Scrabble dictionary. Filter by length 2-15.`
   186→   186→      : `${total > 0 ? `${fmt(total)} valid ` : "All "}words that end with "${letter}" in the official Scrabble dictionary. Filter by length 2-15.`,
   187→   187→  };
   188→   188→}

   369→   369→  }
   370→   370→
   371→   371→  return params;

   430→   430→function isKnownCatchallSlug(slug: string): boolean {
   431→   431→  return (
   432→   432→    slug === "" ||
   433→   433→    (EXTRA_CATCHALL_SLUGS as readonly string[]).includes(slug) ||
   434→   434→    isWordListPage(slug) ||
   435→   435→    Boolean(TOOL_METADATA_REGISTRY[slug])
   436→   436→  );
   437→   437→}

   439→   439→export default async function Page({ params }: PageProps) {
   440→   440→  const resolvedParams = await params;
   441→   441→  const toolSegments = resolvedParams.tool || [];
   442→   442→  const activeToolSlug = toolSegments[0] || "";
   443→   443→  const subRoute = toolSegments[1] || "";
   444→   444→
   445→   445→  // Unknown primary slug → 404 (previously rendered the homepage with 200).
   446→   446→  // "/game" used to alias the blitz page through dead switch-case code; it
   447→   447→  // now redirects to /blitz in next.config.ts. "/unscramble" (homepage
   448→   448→  // alias without a canonical) also redirects to "/" there.
   449→   449→  if (!isKnownCatchallSlug(activeToolSlug)) notFound();
   450→   450→
   451→   451→  // Sub-routes: only the legacy /unscramble/<word-list-slug> pattern is a
   452→   452→  // valid two-segment path (and those are 301-redirected upstream anyway —
   453→   453→  // this is a defensive fallback). Everything else (/wordle/foo, /a/b/c) 404s.
   454→   454→  if (toolSegments.length > 1) {
   455→   455→    const isLegacyWordList =
   456→   456→      activeToolSlug === "unscramble" &&
   457→   457→      toolSegments.length === 2 &&
   458→   458→      isWordListPage(subRoute);
   459→   459→    if (!isLegacyWordList) notFound();
   460→   460→  }
   461→   461→
   462→   462→  let initialWordListData: InitialWordListPage | null = null;
   463→   463→  let topWordData: { word: string; score: number }[] | null = null;
