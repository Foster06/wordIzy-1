    99→function buildFaqJsonLd(toolSlug: string): {
   100→  "@context": string;
   101→  "@type": "FAQPage";
   102→  mainEntity: {
   103→    "@type": "Question";
   104→    name: string;
   105→    acceptedAnswer: { "@type": "Answer"; text: string };
   106→  }[];
   107→} | null {
   108→  // Map route slugs to FAQ keys in translations.en.faq.
   109→  // Most slugs match directly (e.g. "scramble" → faq.scramble).
   110→  // The homepage route is "" but its FAQ is under "unscrambler".
   111→  const faqKey = toolSlug === "" || toolSlug === "unscrambler" ? "unscrambler" : toolSlug;
   112→
   113→  const faqItems = (translations.en.faq as unknown as Record<string, { q: string; a: string }[]>)[faqKey];
   114→  if (!faqItems || faqItems.length === 0) return null;
   115→
   116→  return {
   117→    "@context": "https://schema.org",
   118→    "@type": "FAQPage",
   119→    mainEntity: faqItems.map((item) => ({
   120→      "@type": "Question",
   121→      name: item.q,
   122→      acceptedAnswer: {
   123→        "@type": "Answer",
   124→        text: item.a,
   125→      },
   126→    })),
   127→  };

   143→/** Build the SEO title + description for a programmatic word-list config. */
   144→function buildWordListSeo(config: WordListConfig): { title: string; description: string } {
   145→  const dict = config.dict ?? "scrabble";
   146→  const letter = String(config.value).toUpperCase();
   147→
   148→  // Resolve real word counts from the (module-cached) dictionary store.
   149→  const slug = buildCanonicalWordListUrl(config).replace(/^\//, "");
   150→  const data = getInitialWordListPage(slug, "en", 1);
   151→  const fmt = (n: number) => n.toLocaleString("en-US");
   152→  const total = data?.total ?? 0;
   153→  const count5 = data?.lengthCounts?.[5] ?? 0;
   154→
   155→  if (config.type === "length") {
   156→    const n = config.value as number;
   157→    return {
   158→      title: total > 0
   159→        ? `${n}-Letter Words — ${fmt(total)} Valid Words`
   160→        : `${n}-Letter Words — Complete Scrabble & Wordle List`,
   161→      description: total > 0
   162→        ? `Complete list of ${fmt(total)} ${n}-letter words from the official Scrabble dictionary, sorted by score.`
   163→        : `Complete list of ${n}-letter words from the official Scrabble dictionary, sorted by score.`,
   164→    };
   165→  }
   166→
   167→  const starts = config.type === "starts";
   168→
   169→  if (dict === "wordle") {
   170→    return {
   171→      title: count5 > 0
   172→        ? `${fmt(count5)} 5-Letter Words ${starts ? "Starting" : "Ending"} With "${letter}" — Wordle`
   173→        : `Wordle Words That ${starts ? "Start" : "End"} With "${letter}"`,
   174→      description: starts
   175→        ? `${count5 > 0 ? `${fmt(count5)} five-letter ` : "All "}Wordle words that start with "${letter}", plus 6-7 letter lists. Sorted by score.`
   176→        : `${count5 > 0 ? `${fmt(count5)} five-letter ` : "All "}Wordle words that end with "${letter}". Find hooks, suffixes & patterns fast.`,
   177→    };
   178→  }
   179→
   180→  return {
   181→    title: total > 0
   182→      ? `Words That ${starts ? "Start" : "End"} With "${letter}" — ${fmt(total)} Valid Words`
   183→      : `Words That ${starts ? "Start" : "End"} With "${letter}" — Scrabble Word List`,
   184→    description: starts
   185→      ? `${total > 0 ? `${fmt(total)} valid ` : "All "}words that start with "${letter}" in the official Scrabble dictionary. Filter by length 2-15.`
   186→      : `${total > 0 ? `${fmt(total)} valid ` : "All "}words that end with "${letter}" in the official Scrabble dictionary. Filter by length 2-15.`,
   187→  };
   188→}

   326→export async function generateStaticParams() {
   327→  // Enumerate EVERY valid single-segment slug so the whole site is
   328→  // statically prerendered. Combined with `dynamicParams = false` below,
   329→  // any URL NOT in this list (junk like /nonexistent-page-xyz, /Wordle,
   330→  // /wordle/foo) gets Next.js's built-in REAL 404 — previously the
   331→  // catch-all rendered the homepage for any unknown slug with HTTP 200 +
   332→  // "index, follow", a soft-404 duplicate-content problem that let Google
   333→  // index unlimited homepage copies under arbitrary URLs.
   334→  //
   335→  // Legacy word-list variants (/words-starts-by-*, /words-starting-with-*,
   336→  // /unscramble/<slug>, /N-letter-words) are deliberately NOT enumerated:
   337→  // they are 308-redirected to their canonical URLs in next.config.ts, so
   338→  // they never reach this route.
   339→  const params: { tool: string[] }[] = [];
   340→
   341→  // Homepage
   342→  params.push({ tool: [] });
   343→
   344→  // Every core tool / info page from the metadata registry (unscrambler,
   345→  // scramble, wordle, quordle, scrabble, random, wordfeud, dictionary,
   346→  // wordlists, wordstarts, wordends, wordle-starts, wordle-ends, about,
   347→  // contact, privacy, terms, sitemap, blitz)
   348→  for (const slug of Object.keys(TOOL_METADATA_REGISTRY)) {
   349→    params.push({ tool: [slug] });
   350→  }
   351→
   352→  // Hidden owner-only views
   353→  for (const slug of EXTRA_CATCHALL_SLUGS) {
   354→    params.push({ tool: [slug] });
   355→  }
   356→
   357→  // Programmatic word-list pages:
   358→  // - Scrabble starts/ends A-Z (52)
   359→  // - Wordle starts/ends A-Z (52)
   360→  // - Unscramble by length 2-15 (14)
   361→  for (const letter of ALPHABET) {
   362→    params.push({ tool: [`words-starts-with-${letter}`] });
   363→    params.push({ tool: [`words-ends-with-${letter}`] });
   364→    params.push({ tool: [`wordle-words-starts-with-${letter}`] });
   365→    params.push({ tool: [`wordle-words-ends-with-${letter}`] });
   366→  }
   367→  for (const len of WORD_LIST_LENGTHS) {
   368→    params.push({ tool: [`unscramble-${len}-letter-words`] });
   369→  }
   370→
   371→  return params;
   372→}

   421→// ---------------------------------------------------------------------------
   422→// DEFENSIVE 404 GUARD (2026-11-11).
   423→// The primary guarantee is `dynamicParams = false` above (unknown slugs never
   424→// reach this component). These guards are defense-in-depth in case dynamic
   425→// rendering is ever re-enabled: they keep the old behavior — rendering the
   426→// homepage for ANY unrecognized slug with HTTP 200 "index, follow", a classic
   427→// soft-404 duplicate-content problem — from ever coming back.
   428→// ---------------------------------------------------------------------------
   429→
   430→function isKnownCatchallSlug(slug: string): boolean {
   431→  return (
   432→    slug === "" ||
   433→    (EXTRA_CATCHALL_SLUGS as readonly string[]).includes(slug) ||
   434→    isWordListPage(slug) ||
   435→    Boolean(TOOL_METADATA_REGISTRY[slug])
   436→  );
   437→}
   438→
   439→export default async function Page({ params }: PageProps) {
   440→  const resolvedParams = await params;
   441→  const toolSegments = resolvedParams.tool || [];
   442→  const activeToolSlug = toolSegments[0] || "";
   443→  const subRoute = toolSegments[1] || "";
   444→
   445→  // Unknown primary slug → 404 (previously rendered the homepage with 200).
   446→  // "/game" used to alias the blitz page through dead switch-case code; it
   447→  // now redirects to /blitz in next.config.ts. "/unscramble" (homepage
   448→  // alias without a canonical) also redirects to "/" there.
   449→  if (!isKnownCatchallSlug(activeToolSlug)) notFound();
   450→
   451→  // Sub-routes: only the legacy /unscramble/<word-list-slug> pattern is a
   452→  // valid two-segment path (and those are 301-redirected upstream anyway —
   453→  // this is a defensive fallback). Everything else (/wordle/foo, /a/b/c) 404s.
   454→  if (toolSegments.length > 1) {
   455→    const isLegacyWordList =
   456→      activeToolSlug === "unscramble" &&
   457→      toolSegments.length === 2 &&
   458→      isWordListPage(subRoute);
   459→    if (!isLegacyWordList) notFound();
   460→  }
   461→
   462→  let initialWordListData: InitialWordListPage | null = null;
   463→  let topWordData: { word: string; score: number }[] | null = null;
   464→
   465→  // Try single-segment programmatic slug
   466→  if (isWordListPage(activeToolSlug)) {
   467→    initialWordListData = getInitialWordListPage(activeToolSlug, "en", 50);
   468→    topWordData = getTopScoringWords(activeToolSlug, "en", 10);
   469→  }
   470→  // Try legacy two-segment path: /unscramble/<slug>
   471→  else if (activeToolSlug === "unscramble" && subRoute && isWordListPage(subRoute)) {
   472→    initialWordListData = getInitialWordListPage(subRoute, "en", 50);
   473→    topWordData = getTopScoringWords(subRoute, "en", 10);
   474→  }
   475→
   476→  // Only render the game structured data on the /blitz page
   477→  const isBlitzPage = activeToolSlug === "blitz" || activeToolSlug === "game";
   478→
   479→  // Generic BreadcrumbList for every non-blitz tool page (blitz has its own).
   480→  const isCoreToolPage = Boolean(
   481→    TOOL_METADATA_REGISTRY[activeToolSlug] &&
   482→    !isBlitzPage &&
   483→    activeToolSlug !== "unscrambler" // homepage has its own sitewide WebApplication breadcrumb
   484→  );
   485→  const toolBreadcrumbJsonLd =
   486→    isCoreToolPage && TOOL_METADATA_REGISTRY[activeToolSlug]
   487→      ? buildBreadcrumbJsonLd(
   488→          activeToolSlug,
   489→          TOOL_METADATA_REGISTRY[activeToolSlug].title,
   490→        )
   491→      : null;
   492→
   493→  // FAQPage JSON-LD — for tool pages that have FAQ sections (unscrambler, scramble,
   494→  // wordle, quordle, anagram, random, wordfeud, dictionary, scrabble, wordlists,
   495→  // wordstarts, wordends). Gets rich snippets in Bing/Google search results.
   496→  // Not rendered for blitz, info pages, or word-list programmatic pages.
   497→  const faqJsonLd = isCoreToolPage || activeToolSlug === "" || activeToolSlug === "unscrambler"
   498→    ? buildFaqJsonLd(activeToolSlug)
   499→    : null;
   500→
   501→  return (
   502→    <>
   503→      {isBlitzPage && (
   504→        <script
   505→          type="application/ld+json"
   506→          dangerouslySetInnerHTML={{ __html: JSON.stringify(blitzGameJsonLd) }}
   507→        />
   508→      )}
   509→      {isBlitzPage && (
   510→        <script
   511→          type="application/ld+json"
   512→          dangerouslySetInnerHTML={{ __html: JSON.stringify(blitzBreadcrumbJsonLd) }}
   513→        />
   514→      )}
   515→      {toolBreadcrumbJsonLd && (
   516→        <script
   517→          type="application/ld+json"
   518→          dangerouslySetInnerHTML={{ __html: JSON.stringify(toolBreadcrumbJsonLd) }}
   519→        />
   520→      )}
   521→      {faqJsonLd && (
   522→        <script
   523→          type="application/ld+json"
   524→          dangerouslySetInnerHTML={{ __html: JSON.stringify(faqJsonLd) }}
   525→        />
   526→      )}
   527→      <MainToolView
   528→        params={params}
   529→        initialWordListData={initialWordListData}
   530→        topWordData={topWordData}
   531→      />
   532→    </>
   533→  );
   534→}
