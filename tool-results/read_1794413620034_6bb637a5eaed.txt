    20→/** Length of a string AFTER HTML attribute escaping, matching what Next.js
    21→ *  renders in `<meta content="…">`. Escapable characters expand:
    22→ *  & → &amp; (5), < → &lt; (4), > → &gt; (4), " → &quot; (6),
    23→ *  ' → &#x27; (6). Used to keep the meta description within Bing's
    24→ *  25-160 char limit as it appears in the final HTML. */
    25→function escapedLength(s: string): number {
    26→  let n = 0;
    27→  for (const ch of s) {
    28→    if (ch === '"') n += 6; // &quot;
    29→    else if (ch === "'") n += 6; // &#x27;
    30→    else if (ch === "&") n += 5; // &amp;
    31→    else if (ch === "<" || ch === ">") n += 4; // &lt; / &gt;
    32→    else n += 1;
    33→  }
    34→  return n;
    35→}

    38→// Each word page reports validity across 8 dictionaries (2 English + 6

    42→function joinList(parts: string[]): string {
    43→  if (parts.length === 0) return "";
    44→  if (parts.length === 1) return parts[0];
    45→  return `${parts.slice(0, -1).join(", ")} and ${parts[parts.length - 1]}`;
    46→}

    58→/** "French, Dutch and 2 more" — compact list for the meta description. */
    59→function otherLangsShort(data: ScrabbleWordData): string {

    81→/** Generate per-word metadata tuned for the "is X a scrabble word" SERP. */
    82→export async function generateMetadata({ params }: PageProps): Promise<Metadata> {
    83→  const { word: rawWord } = await params;
    84→  const word = rawWord.toLowerCase();
    85→  if (!/^[a-z]{2,15}$/.test(word)) return {};
    86→
    87→  const data = getScrabbleWordData(word);
    88→  const display = data.display;
    89→  const otherCount = data.validInOtherLanguages.length;
    90→  const title = data.valid
    91→    ? `Is ${display} a Valid Scrabble Word? Yes — ${data.points} Points`
    92→    : otherCount > 0
    93→      ? `Is ${display} a Valid Scrabble Word? Not in English`
    94→      : `Is ${display} a Valid Scrabble Word? No — Not in NWL or CSW`;
    95→
    96→  // Meta description MUST be 25-160 chars (Bing URL Inspection rule).
    97→  // Target 150 to leave a safety buffer: Next.js HTML-escapes the content
    98→  // attribute (& → &amp;, < → &lt;, ' → &#x27;, …), which EXPANDS the
    99→  // rendered length beyond the raw string length. The budget below is
   100→  // computed on the ESCAPED length so the rendered attribute never
   101→  // exceeds the target.
   102→  const MAX_DESC = 150;
   103→  let description: string;
   104→  if (data.valid) {
   105→    const dicts = `${data.inNWL ? "NWL2023" : ""}${data.inNWL && data.inCSW ? " & " : ""}${data.inCSW ? "CSW21" : ""}`;
   106→    const fixed = `${display} is a valid Scrabble word worth ${data.points} points. Valid in ${dicts}.`;
   107→    let desc = fixed;
   108→    if (data.definition) {
   109→      const budget = MAX_DESC - escapedLength(fixed) - " Definition: ".length;
   110→      if (budget >= 20) {
   111→        let defPart = data.definition.slice(0, budget);
   112→        while (escapedLength(`${fixed} Definition: ${defPart}`) > MAX_DESC && defPart.length > 20) {
   113→          defPart = defPart.slice(0, -1);
   114→        }
   115→        desc = `${fixed} Definition: ${defPart}`;
   116→      }
   117→    }
   118→    description = desc;
   119→  } else if (otherCount > 0) {
   120→    description = `${display} is not a valid English Scrabble word, but it is valid in ${otherLangsShort(data)}. See why, plus similar valid English words.`;
   121→  } else {
   122→    description = `${display} is not a valid Scrabble word in NWL2023 or CSW21. See why, and find similar valid words to play.`;
   123→  }
   124→
   125→  const canonical = `${SITE_ORIGIN}/scrabble-words/${word}`;
   126→
   127→  // FAQPage JSON-LD — embedded as <meta> via `other` field, rendered as a script
   128→  // tag in the page body. Three Q&As that match common SERP feature triggers.
   129→  const faqJsonLd = buildFaqJsonLd(data);
   130→
   131→  return {
   132→    title,
   133→    description,
   134→    alternates: { canonical },
   135→    openGraph: {
   136→      title,
   137→      description,
   138→      url: canonical,
   139→      siteName: "wordIzy",
   140→      type: "article",
   141→      images: [{ url: "/og-image.png", width: 1200, height: 630, alt: title }],
   142→    },
   143→    twitter: {
   144→      card: "summary_large_image",
   145→      title,
   146→      description,
   147→      images: ["/og-image.png"],
   148→    },
   149→    other: faqJsonLd ? { "ld:faq": JSON.stringify(faqJsonLd) } : undefined,
   150→  };
   151→}

   153→/** Build a 3-question FAQPage JSON-LD schema for the word.
   154→ *  Unlocks expandable Q&A rich snippets in Google/Bing SERPs. */
   155→function buildFaqJsonLd(data: ScrabbleWordData) {
   156→  const display = data.display;
   157→  const faqItems: { q: string; a: string }[] = [];
   158→
   159→  if (data.valid) {
   160→    faqItems.push({
   161→      q: `Is ${display} a valid Scrabble word?`,
   162→      a: `Yes, ${display} is a valid Scrabble word${data.inNWL ? " in the NWL2023 (Tournament Word List, the North American official dictionary)" : ""}${data.inNWL && data.inCSW ? " and" : data.inCSW ? " in the" : ""}${data.inCSW ? " the CSW21 (Collins Scrabble Words, the international dictionary)" : ""}. It is worth ${data.points} Scrabble points.`,
   163→    });
   164→    if (data.definition) {
   165→      faqItems.push({
   166→        q: `What does ${display} mean in Scrabble?`,
   167→        a: `${display} is ${expandPOS(data.partOfSpeech)} as: ${data.definition}.`,
   168→      });
   169→    }
   170→    faqItems.push({
   171→      q: `How many points is ${display} worth in Scrabble?`,
   172→      a: `${display} is worth ${data.points} points in Scrabble, calculated as the sum of each letter's tile value: ${data.letterBreakdown
   173→        .map((l) => `${l.letter}=${l.value}${l.count > 1 ? ` (×${l.count})` : ""}`)
   174→        .join(", ")}.`,
   175→    });
   176→  } else {
   177→    faqItems.push({
   178→      q: `Is ${display} a valid Scrabble word?`,
   179→      a: `No, ${display} is not a valid Scrabble word in either the NWL2023 (Tournament Word List) or CSW21 (Collins Scrabble Words) dictionary. It cannot be played in an official English-language Scrabble game${data.validInOtherLanguages.length > 0 ? ` — but it is a valid word in ${otherLangsFull(data)}` : ""}.`,
   180→    });
   181→    faqItems.push({
   182→      q: `Why is ${display} not a valid Scrabble word?`,
   183→      a: `${display} does not appear in the official English Scrabble dictionaries (NWL2023 for North America, CSW21 for international play). English-language Scrabble only accepts words that are listed in these dictionaries. Common reasons a word is excluded: it's a proper noun (capitalized), an abbreviation, a prefix/suffix, or simply not recognized as a standard English word.`,
   184→    });
   185→    faqItems.push({
   186→      q: `Where can I check if a word is valid in Scrabble?`,
   187→      a: `You can use the wordIzy Scrabble Word Checker at wordizy.com/scrabble to look up any word. This page also shows the word's point value, definition, and letter breakdown if it is valid.`,
   188→    });
   189→  }
   190→
   191→  // Cross-language FAQ — added whenever the word is valid in at least one
   192→  // other language's dictionary (works for valid and invalid English words).
   193→  if (data.validInOtherLanguages.length > 0) {
   194→    faqItems.push({
   195→      q: `Is ${display} a valid word in other languages?`,
   196→      a: data.valid
   197→        ? `Yes — besides English, ${display} is also a valid word in ${otherLangsFull(data)}.`
   198→        : `Yes. ${display} is a valid word in ${otherLangsFull(data)}. It is not valid in English-language Scrabble (NWL2023 or CSW21), but it can be played in Scrabble games that use those languages' dictionaries.`,
   199→    });
   200→  }
   201→
   202→  return {
   203→    "@context": "https://schema.org",
   204→    "@type": "FAQPage",
   205→    mainEntity: faqItems.map((item) => ({
   206→      "@type": "Question",
   207→      name: item.q,
   208→      acceptedAnswer: { "@type": "Answer", text: item.a },
   209→    })),
   210→  };
   211→}

   213→/** Expand a part-of-speech code (n, v, adj, adv, interj) into a readable phrase. */
   214→function expandPOS(pos: string | null): string {
   215→  if (!pos) return "defined";
   216→  const map: Record<string, string> = {
   217→    n: "a noun",
   218→    v: "a verb",
   219→    adj: "an adjective",
   220→    adv: "an adverb",
   221→    interj: "an interjection",
   222→    article: "an article",
   223→    prep: "a preposition",
   224→    pron: "a pronoun",
   225→    conj: "a conjunction",
   226→  };
   227→  return map[pos] || "defined";
   228→}

   452→              Frequently Asked Questions
   453→            </h2>
   454→            <div className="space-y-4">
   455→              {data.valid ? (
   456→                <>
   457→                  <FaqItem q={`Is ${data.display} a valid Scrabble word?`}>
   458→                    Yes, {data.display} is a valid Scrabble word
   459→                    {data.inNWL ? " in the NWL2023 (Tournament Word List, the North American official dictionary)" : ""}
   460→                    {data.inNWL && data.inCSW ? " and" : data.inCSW ? " in the" : ""}
   461→                    {data.inCSW ? " the CSW21 (Collins Scrabble Words, the international dictionary)" : ""}.
   462→                    It is worth {data.points} Scrabble points.
   463→                  </FaqItem>
   464→                  {data.definition && (
   465→                    <FaqItem q={`What does ${data.display} mean in Scrabble?`}>
   466→                      {data.display} is {expandPOS(data.partOfSpeech)} as: {data.definition}.
   467→                    </FaqItem>
   468→                  )}
   469→                  <FaqItem q={`How many points is ${data.display} worth in Scrabble?`}>
   470→                    {data.display} is worth {data.points} points in Scrabble, calculated as the sum of each
   471→                    letter&apos;s tile value:{" "}
   472→                    {data.letterBreakdown
   473→                      .map((l) => `${l.letter}=${l.value}${l.count > 1 ? ` (×${l.count})` : ""}`)
   474→                      .join(", ")}.
   475→                  </FaqItem>
   476→                </>
   477→              ) : (
   478→                <>
   479→                  <FaqItem q={`Is ${data.display} a valid Scrabble word?`}>
   480→                    No, {data.display} is not a valid Scrabble word in either the NWL2023 (Tournament
   481→                    Word List) or CSW21 (Collins Scrabble Words) dictionary. It cannot be played in
   482→                    an official English-language Scrabble game
   483→                    {data.validInOtherLanguages.length > 0
   484→                      ? ` — but it is a valid word in ${otherLangsFull(data)}`
   485→                      : ""}
   486→                    .
   487→                  </FaqItem>
   488→                  <FaqItem q={`Why is ${data.display} not a valid Scrabble word?`}>
   489→                    {data.display} does not appear in the official English Scrabble dictionaries (NWL2023 for
   490→                    North America, CSW21 for international play). English-language Scrabble only accepts words
   491→                    that are listed in these dictionaries. Common reasons a word is excluded: it&apos;s a
   492→                    proper noun (capitalized), an abbreviation, a prefix/suffix, or simply not
   493→                    recognized as a standard English word.
   494→                  </FaqItem>
   495→                  <FaqItem q="Where can I check if a word is valid in Scrabble?">
   496→                    You can use the wordIzy Scrabble Word Checker at{" "}
   497→                    <Link href="/scrabble" className="text-brand underline underline-offset-2">wordizy.com/scrabble</Link>{" "}
   498→                    to look up any word. This page also shows the word&apos;s point value, definition,
   499→                    and letter breakdown if it is valid.
   500→                  </FaqItem>
   501→                </>
   502→              )}
   503→              {data.validInOtherLanguages.length > 0 && (
   504→                <FaqItem q={`Is ${data.display} a valid word in other languages?`}>
   505→                  {data.valid
   506→                    ? `Yes — besides English, ${data.display} is also a valid word in ${otherLangsFull(data)}.`
   507→                    : `Yes. ${data.display} is a valid word in ${otherLangsFull(data)}. It is not valid in English-language Scrabble (NWL2023 or CSW21), but it can be played in Scrabble games that use those languages' dictionaries.`}
   508→                </FaqItem>
   509→              )}
   510→            </div>
   511→          </section>
   512→
   513→          {/* FAQPage JSON-LD schema */}
   514→          {faqJsonLd && (
   515→            <script
   516→              type="application/ld+json"
   517→              dangerouslySetInnerHTML={{ __html: JSON.stringify(faqJsonLd) }}
   518→            />
   519→          )}
   520→
   521→          {/* BreadcrumbList JSON-LD */}
   522→          <script
   523→            type="application/ld+json"
   524→            dangerouslySetInnerHTML={{
   525→              __html: JSON.stringify({
   526→                "@context": "https://schema.org",
   527→                "@type": "BreadcrumbList",
   528→                itemListElement: [
   529→                  { "@type": "ListItem", position: 1, name: "wordIzy", item: SITE_ORIGIN },
   530→                  { "@type": "ListItem", position: 2, name: "Scrabble Words", item: `${SITE_ORIGIN}/scrabble-words` },
   531→                  { "@type": "ListItem", position: 3, name: `Is ${data.display} a Scrabble Word?`, item: `${SITE_ORIGIN}/scrabble-words/${word}` },
   532→                ],
   533→              }),
   534→            }}
   535→          />
   536→
   537→          <ShareButtons
   538→            url={`${SITE_ORIGIN}/scrabble-words/${word}`}
   539→            title={`Is ${data.display} a Valid Scrabble Word?`}
   540→          />
   541→
   542→          {/* Internal links */}
   543→          <div className="mt-10 pt-6 border-t border-white/10">
   544→            <h3 className="text-sm font-semibold text-foreground mb-3">Related Tools</h3>
   545→            <div className="flex flex-wrap gap-2">
   546→              <Link href="/scrabble" className="px-3 py-1.5 rounded-md glass-soft text-xs text-foreground/80 hover:text-brand transition-colors">
   547→                Scrabble Word Checker
   548→              </Link>
   549→              <Link href={`/words-starts-with-${word[0]}`} className="px-3 py-1.5 rounded-md glass-soft text-xs text-foreground/80 hover:text-brand transition-colors">
   550→                Words Starting With {word[0].toUpperCase()}
   551→              </Link>
   552→              <Link href={`/unscramble-${data.length}-letter-words`} className="px-3 py-1.5 rounded-md glass-soft text-xs text-foreground/80 hover:text-brand transition-colors">
   553→                {data.length}-Letter Words
   554→              </Link>
   555→              <Link href="/wordlists" className="px-3 py-1.5 rounded-md glass-soft text-xs text-foreground/80 hover:text-brand transition-colors">
   556→                All Word Lists
   557→              </Link>
   558→              <Link href="/dictionary" className="px-3 py-1.5 rounded-md glass-soft text-xs text-foreground/80 hover:text-brand transition-colors">
   559→                Dictionary
   560→              </Link>
   561→            </div>
   562→          </div>
   563→        </article>
   564→      </main>
   565→      <SiteFooter />
   566→    </div>
   567→  );
   568→}
